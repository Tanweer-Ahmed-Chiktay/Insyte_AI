generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
  directUrl = env("DIRECT_DATABASE_URL")
}

model User {
  id            String          @id @default(cuid())
  email         String          @unique
  name          String?
  image         String?
  emailVerified DateTime?       @map("email_verified")
  createdAt     DateTime        @default(now()) @map("created_at")
  updatedAt     DateTime        @updatedAt @map("updated_at")
  historyId     String?         @map("history_id")
  
  // Relations
  accounts      Account[]
  sessions      Session[]
  emails        Email[]
  gmailCredentials GmailCredentials[]
  gmailSyncStatus  GmailSyncStatus[]

  @@map("users")
}

model Account {
  id                String  @id @default(cuid())
  userId            String  @map("user_id")
  type              String
  provider          String
  providerAccountId String  @map("provider_account_id")
  refresh_token     String? @db.Text
  access_token      String? @db.Text
  expires_at        Int?
  token_type        String?
  scope             String?
  id_token          String? @db.Text
  session_state     String?
  createdAt         DateTime @default(now()) @map("created_at")
  updatedAt         DateTime @updatedAt @map("updated_at")

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([provider, providerAccountId])
  @@map("accounts")
}

model Session {
  id           String   @id @default(cuid())
  sessionToken String   @unique @map("session_token")
  userId       String   @map("user_id")
  expires      DateTime
  createdAt    DateTime @default(now()) @map("created_at")
  updatedAt    DateTime @updatedAt @map("updated_at")

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@map("sessions")
}

model GmailCredentials {
  id                    String    @id @default(cuid())
  userId                String    @map("user_id")
  email                 String
  encryptedRefreshToken String    @map("encrypted_refresh_token") @db.Text
  encryptedAccessToken  String?   @map("encrypted_access_token") @db.Text
  tokenExpiresAt        DateTime? @map("token_expires_at")
  scope                 String?
  watchExpiration       DateTime? @map("watch_expiration")
  historyId             String?   @map("history_id")
  isActive              Boolean   @default(true) @map("is_active")
  createdAt             DateTime  @default(now()) @map("created_at")
  updatedAt             DateTime  @updatedAt @map("updated_at")

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([userId, email])
  @@map("gmail_credentials")
}

model Email {
  id              String      @id @default(cuid())
  userId          String      @map("user_id")
  gmailId         String      @map("gmail_id")
  threadId        String      @map("thread_id")
  subject         String?     @db.Text
  snippet         String?     @db.Text
  senderName      String?     @map("sender_name")
  senderEmail     String?     @map("sender_email")
  recipientEmails String[]    @map("recipient_emails")
  receivedAt      DateTime    @map("date_received")
  isRead          Boolean     @default(false) @map("is_read")
  isStarred       Boolean     @default(false) @map("is_starred")
  isImportant     Boolean     @default(false) @map("is_important")
  isDeleted       Boolean     @default(false) @map("is_deleted")
  labelIds        String[]    @map("labels")
  hasAttachments  Boolean     @default(false) @map("has_attachments")
  bodyText        String?     @map("body_text") @db.Text
  bodyHtml        String?     @map("body_html") @db.Text
  rawHeaders      Json?       @map("raw_headers")
  createdAt       DateTime    @default(now()) @map("created_at")
  updatedAt       DateTime    @updatedAt @map("updated_at")

  // Relations
  user         User           @relation(fields: [userId], references: [id], onDelete: Cascade)
  attachments  GmailAttachment[]

  // Backward compatibility fields
  from         String         @default("") // Maps to senderEmail
  to           String         @default("") // Maps to recipientEmails[0]

  @@unique([userId, gmailId])
  @@map("gmail_messages")
}

model GmailThread {
  id              String    @id @default(cuid())
  userId          String    @map("user_id")
  gmailThreadId   String    @map("gmail_thread_id")
  subject         String?   @db.Text
  snippet         String?   @db.Text
  messageCount    Int       @default(1) @map("message_count")
  unreadCount     Int       @default(0) @map("unread_count")
  lastMessageDate DateTime? @map("last_message_date")
  participants    String[]
  labelIds        String[]  @map("labels")
  isDeleted       Boolean   @default(false) @map("is_deleted")
  createdAt       DateTime  @default(now()) @map("created_at")
  updatedAt       DateTime  @updatedAt @map("updated_at")

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([userId, gmailThreadId])
  @@map("gmail_threads")
}

model GmailAttachment {
  id                String   @id @default(cuid())
  messageId         String   @map("message_id")
  gmailAttachmentId String   @map("gmail_attachment_id")
  filename          String?
  mimeType          String?  @map("mime_type")
  sizeBytes         BigInt?  @map("size_bytes")
  contentId         String?  @map("content_id")
  isInline          Boolean  @default(false) @map("is_inline")
  createdAt         DateTime @default(now()) @map("created_at")

  message Email @relation(fields: [messageId], references: [id], onDelete: Cascade)

  @@unique([messageId, gmailAttachmentId])
  @@map("gmail_attachments")
}

model GmailSyncStatus {
  id             String    @id @default(cuid())
  userId         String    @map("user_id")
  email          String
  lastSyncAt     DateTime? @map("last_sync_at")
  lastHistoryId  String?   @map("last_history_id")
  syncStatus     String    @default("pending") @map("sync_status") // pending, syncing, completed, error
  errorMessage   String?   @map("error_message") @db.Text
  totalMessages  Int       @default(0) @map("total_messages")
  syncedMessages Int       @default(0) @map("synced_messages")
  createdAt      DateTime  @default(now()) @map("created_at")
  updatedAt      DateTime  @updatedAt @map("updated_at")

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([userId, email])
  @@map("gmail_sync_status")
}
